def ref_grid(value, width):
    """Independent reference: round on the global subnormal grid by distance comparison."""
    if width == 32: p, emin, emax, bias = 24, -126, 127, 127
    else: p, emin, emax, bias = 53, -1022, 1023, 1023
    sign = 1 if value < 0 or (value == 0 and 0) else 0
    v = abs(value)
    if v == 0: return 0, False
    q = F(2) ** (emin - (p - 1))          # subnormal grid spacing
    k = v / q
    lo = k.numerator // k.denominator
    hi = lo + 1
    dlo, dhi = k - lo, hi - k
    if dlo < dhi: sig = lo

def ref(value, width):
    """Reference: locate binade, quantize by floor/ceil + exact distance, ties to even."""
    if width == 32: p, emin, emax, bias = 24, -126, 127, 127
    else: p, emin, emax, bias = 53, -1022, 1023, 1023
    sign = 1 if value < 0 else 0
    v = abs(value)
    if v == 0: return 0, False
    e = v.numerator.bit_length() - v.denominator.bit_length()
    if F(2) ** e > v: e -= 1
    assert F(2) ** e <= v < F(2) ** (e + 1)
    s = F(2) ** (max(e, emin) - (p - 1))     # target spacing
    k = v / s
    lo = k.numerator // k.denominator
    hi = lo + 1
    if k - lo < hi - k: m = lo
    elif hi - k < k - lo: m = hi
    else: m = lo if lo % 2 == 0 else hi
    if m == 1 << p:                            # carried into next binade
        m >>= 1; e += 1
    if e > emax: return None, True             # -> inf
    if m < 1 << (p - 1): return (sign << (width - 1)) | m, False
    return (sign << (width - 1)) | ((e + bias) << (p - 1)) | (m - (1 << (p - 1))), False
